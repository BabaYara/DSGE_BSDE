\begin{lemma}[Share dynamics]\label{lem:s}

The share process obeys $\mathrm{d}s_i=\mu_{s,i}(\mathbf s)\,\mathrm{d}t+\sigma_{s,i}(\mathbf s)^\top \mathrm{d}\mathbf W_t$.
\end{lemma}

\begin{proposition}[SDF]\label{prop:sdf}
\lean{LucasNotes.sdf_drift_algebra}
\leanok
\uses{lem:C}
With log utility, $\Lambda_t=e^{-\rho t} C_t^{-1}$ and $\lambda_t=\sigma_C(\mathbf s_t)$. The algebraic drift identity is checked by
the Lean theorem `LucasNotes.sdf_drift_algebra` (cancelling denominators with $C\neq 0$), used in the It\^o derivation.
\end{proposition}

\begin{theorem}[Valuation PDE]\label{thm:valuation_pde_reduced}
\leanok

$\mathcal L^{\mathbb Q}_{\mathbf s} g_i + \mathcal C_{\mathbf s} g_i - \rho g_i + s_i = 0$ on $\Delta^{N-1}$.
\end{theorem}
